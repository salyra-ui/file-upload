cmake_minimum_required(VERSION 3.21)
project(salyra_upload VERSION 0.1.0 LANGUAGES C CXX)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CJSON REQUIRED IMPORTED_TARGET libcjson)
add_library(salyra_upload src/engine.c src/filesystem.c)
target_include_directories(salyra_upload PUBLIC include)
target_link_libraries(salyra_upload PUBLIC OpenSSL::Crypto PkgConfig::CJSON m)
target_compile_features(salyra_upload PUBLIC c_std_11)
target_compile_options(salyra_upload PRIVATE -Wall -Wextra -Wpedantic)
option(SALYRA_UPLOAD_EXAMPLES "Build the bounded local HTTP example" ${PROJECT_IS_TOP_LEVEL})
option(SALYRA_UPLOAD_TESTS "Build C and C++ ownership tests" ${PROJECT_IS_TOP_LEVEL})
if(SALYRA_UPLOAD_EXAMPLES)
  add_executable(salyra_upload_example examples/http.c)
  target_link_libraries(salyra_upload_example PRIVATE salyra_upload)
endif()
if(SALYRA_UPLOAD_TESTS)
  enable_testing()
  add_executable(salyra_upload_test tests/core.c)
  target_link_libraries(salyra_upload_test PRIVATE salyra_upload)
  target_compile_options(salyra_upload_test PRIVATE -UNDEBUG)
  add_executable(salyra_upload_cpp_test tests/raii.cpp)
  target_link_libraries(salyra_upload_cpp_test PRIVATE salyra_upload)
  target_compile_features(salyra_upload_cpp_test PRIVATE cxx_std_17)
  target_compile_options(salyra_upload_cpp_test PRIVATE -UNDEBUG)
  add_test(NAME core COMMAND salyra_upload_test)
  add_test(NAME cpp_ownership COMMAND salyra_upload_cpp_test)
endif()
include(GNUInstallDirs)
install(TARGETS salyra_upload ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/salyra_upload.h include/salyra_upload.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
